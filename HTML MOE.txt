  <script>
    // 16 Division Financial Master Dataset (Replace values with exact IFMIS data)
    const moeData = [
      { code: "DIV-01", name: "HQ & General Administration", budget: 1500000, actual: 850000, commitments: 120000 },
      { code: "DIV-02", name: "Accounts & Financial Management", budget: 600000, actual: 410000, commitments: 30000 },
      { code: "DIV-03", name: "HR & Personnel Services", budget: 500000, actual: 320000, commitments: 15000 },
      { code: "DIV-04", name: "Policy, Planning & Research", budget: 450000, actual: 210000, commitments: 40000 },
      { code: "DIV-05", name: "Primary Education Division", budget: 8500000, actual: 6200000, commitments: 450000 },
      { code: "DIV-06", name: "Junior Secondary Schools (JSS)", budget: 6200000, actual: 4800000, commitments: 310000 },
      { code: "DIV-07", name: "Senior Secondary Schools (SSS)", budget: 7800000, actual: 5900000, commitments: 620000 },
      { code: "DIV-08", name: "School Improvement & Curriculum", budget: 950000, actual: 610000, commitments: 85000 },
      { code: "DIV-09", name: "Inclusive & Special Education", budget: 400000, actual: 280000, commitments: 20000 },
      { code: "DIV-10", name: "Early Childhood Education (ECE)", budget: 1200000, actual: 780000, commitments: 95000 },
      { code: "DIV-11", name: "Teacher Development & Training", budget: 1100000, actual: 890000, commitments: 110000 },
      { code: "DIV-12", name: "Examinations & Assessment", budget: 750000, actual: 640000, commitments: 50000 },
      { code: "DIV-13", name: "Assets & Infrastructure Maintenance", budget: 2300000, actual: 1100000, commitments: 750000 },
      { code: "DIV-14", name: "IT & Education Management (EMIS)", budget: 800000, actual: 490000, commitments: 130000 },
      { code: "DIV-15", name: "Outer Islands Educational Services", budget: 3100000, actual: 2150000, commitments: 280000 },
      { code: "DIV-16", name: "Technical & Vocational Grants", budget: 1800000, actual: 1400000, commitments: 200000 }
    ];

    function fmt(val) {
      return '$' + val.toLocaleString();
    }

    // Compute Totals
    let totBudget = 0, totActual = 0, totComm = 0;
    moeData.forEach(d => {
      totBudget += d.budget;
      totActual += d.actual;
      totComm += d.commitments;
    });

    const totBalance = totBudget - (totActual + totComm);
    const totBurn = ((totActual + totComm) / totBudget) * 100;

    document.getElementById('total-budget').innerText = fmt(totBudget);
    document.getElementById('total-actual').innerText = fmt(totActual);
    document.getElementById('total-commitments').innerText = fmt(totComm);
    document.getElementById('total-balance').innerText = fmt(totBalance);
    document.getElementById('total-burn').innerText = totBurn.toFixed(1) + '%';

    // Populate Table
    const tbody = document.getElementById('table-body');
    moeData.forEach(d => {
      const balance = d.budget - (d.actual + d.commitments);
      const burn = ((d.actual + d.commitments) / d.budget) * 100;
      let badgeClass = "status-good", statusTxt = "On Track";

      if (burn > 90) { badgeClass = "status-danger"; statusTxt = "High Risk"; }
      else if (burn < 50) { badgeClass = "status-warn"; statusTxt = "Under-spent"; }

      const tr = document.createElement('tr');
      tr.innerHTML = `
        <td><b>${d.code}</b></td>
        <td>${d.name}</td>
        <td>${fmt(d.budget)}</td>
        <td>${fmt(d.actual)}</td>
        <td>${fmt(d.commitments)}</td>
        <td>${fmt(balance)}</td>
        <td>${burn.toFixed(1)}%</td>
        <td><span class="status-badge ${badgeClass}">${statusTxt}</span></td>
      `;
      tbody.appendChild(tr);
    });

    // Bar Chart
    new Chart(document.getElementById('barChart'), {
      type: 'bar',
      data: {
        labels: moeData.map(d => d.code),
        datasets: [
          { label: 'Actual Spend', data: moeData.map(d => d.actual), backgroundColor: '#003366' },
          { label: 'Commitments', data: moeData.map(d => d.commitments), backgroundColor: '#ff9900' },
          { label: 'Remaining Balance', data: moeData.map(d => d.budget - (d.actual + d.commitments)), backgroundColor: '#e0e0e0' }
        ]
      },
      options: { responsive: true, scales: { x: { stacked: true }, y: { stacked: true } } }
    });

    // Doughnut Chart
    new Chart(document.getElementById('doughnutChart'), {
      type: 'doughnut',
      data: {
        labels: ['Actual Spend', 'Commitments', 'Remaining Balance'],
        datasets: [{
          data: [totActual, totComm, totBalance],
          backgroundColor: ['#003366', '#ff9900', '#e0e0e0']
        }]
      },
      options: { responsive: true }
    });
  </script>
